/* =========================
   CODE ANALYSIS
========================= */

function analyzeCode() {

    const code = document.getElementById("codeInput").value.trim();

    if (!code) {
        showToast("Please enter some code first");
        return;
    }

    currentCode = code;

    const language =
        document.getElementById("language").value;

    document.getElementById("detectedLanguage").textContent =
        language.toUpperCase();

    let complexity = "O(1)";

    if (/for|while|foreach/.test(code)) {
        complexity = "O(n)";
    }

    if (
        /for[\s\S]*for/.test(code) ||
        /while[\s\S]*while/.test(code)
    ) {
        complexity = "O(n²)";
    }

    document.getElementById("complexity").textContent =
        complexity;

    const list = document.getElementById("analysisList");

    list.innerHTML = "";

    const features = [];

    if (/input|scanf|cin|prompt|read/i.test(code)) {
        features.push("📥 User input operation detected");
    }

    if (/for|while|foreach/i.test(code)) {
        features.push("🔁 Loop structure detected");
    }

    if (/if|else|switch|case/i.test(code)) {
        features.push("🔀 Conditional decision detected");
    }

    if (/function|def |void |int main|public static void/i.test(code)) {
        features.push("⚙️ Function / method detected");
    }

    if (/print|printf|cout|console\.log|println/i.test(code)) {
        features.push("📤 Output operation detected");
    }

    if (/[+\-*\/%=]/.test(code)) {
        features.push("➗ Arithmetic operation detected");
    }

    if (/class /i.test(code)) {
        features.push("🏛 Class / object-oriented structure detected");
    }

    if (/array|\[.*\]/i.test(code)) {
        features.push("📦 Array / collection structure detected");
    }

    if (features.length === 0) {
        features.push("🔍 Basic program structure detected");
    }

    features.forEach(feature => {

        const li = document.createElement("li");

        li.textContent = feature;

        list.appendChild(li);
    });

    buildFlow(code);

    showPage(
        "analysis",
        document.querySelector('.nav-item[onclick*="analysis"]')
    );

    showToast("AI analysis completed");
}


/* =========================
   EXECUTION FLOW
========================= */

function buildFlow(code) {

    const flow = document.getElementById("flow");

    let nodes = [];

    if (/input|scanf|cin|prompt|read/i.test(code)) {
        nodes.push("📥 Input");
    }

    nodes.push("⚙️ Processing");

    if (/if|else|switch|case/i.test(code)) {
        nodes.push("🔀 Decision");
    }

    if (/for|while|foreach/i.test(code)) {
        nodes.push("🔁 Loop");
    }

    if (/function|def |void |int main|public static void/i.test(code)) {
        nodes.push("⚙️ Function");
    }

    if (/print|printf|cout|console\.log|println/i.test(code)) {
        nodes.push("📤 Output");
    }

    flow.innerHTML = "";

    nodes.forEach((node,index) => {

        const div = document.createElement("div");

        div.className = "flow-node";

        div.textContent = node;

        flow.appendChild(div);

        if (index < nodes.length - 1) {

            const arrow = document.createElement("div");

            arrow.className = "arrow";

            arrow.textContent = "→";

            flow.appendChild(arrow);
        }
    });
}

function startQuiz() {

    quizStarted = true;

    quizIndex = 0;

    userAnswers =
        new Array(quizQuestions.length).fill(null);

    document.getElementById("quizCard").style.display = "block";

    document.getElementById("scoreResult").style.display = "none";

    renderQuestion();
}


/* =========================
   RENDER QUIZ
========================= */

function renderQuestion() {

    const q = quizQuestions[quizIndex];

    document.getElementById("questionNumber").textContent =
        `Question ${quizIndex + 1} of ${quizQuestions.length}`;

    document.getElementById("questionType").textContent =
        q.type;

    document.getElementById("questionText").textContent =
        q.question;

    const codeBox =
        document.getElementById("questionCode");

    if (q.code) {

        codeBox.style.display = "block";

        codeBox.textContent = q.code;

    } else {

        codeBox.style.display = "none";
    }

    const options =
        document.getElementById("options");

    options.innerHTML = "";

    q.options.forEach((option,index) => {

        const button =
            document.createElement("button");

        button.className = "option";

        button.textContent =
            `${String.fromCharCode(65 + index)}. ${option}`;

        button.onclick =
            () => selectAnswer(index);

        options.appendChild(button);

    });

    const feedback =
        document.getElementById("feedback");

    feedback.className = "answer-feedback";

    feedback.innerHTML = "";

    const existing =
        userAnswers[quizIndex];

    if (existing !== null) {

        showAnswer(existing,false);
    }

    updateQuizStats();

    const percentage =
        ((quizIndex + 1) / quizQuestions.length) * 100;

    document.getElementById("progressBar").style.width =
        percentage + "%";

    document.getElementById("previousBtn").disabled =
        quizIndex === 0;

    document.getElementById("nextBtn").textContent =
        quizIndex === quizQuestions.length - 1
            ? "Finish Quiz"
            : "Next →";
}


/* =========================
   SELECT ANSWER
========================= */

function selectAnswer(index) {

    if (userAnswers[quizIndex] !== null) {
        return;
    }

    userAnswers[quizIndex] = index;

    showAnswer(index,true);

    updateQuizStats();
}


/* =========================
   SHOW ANSWER
========================= */

function showAnswer(selected,scroll) {

    const q = quizQuestions[quizIndex];

    const optionButtons =
        document.querySelectorAll(".option");

    optionButtons.forEach((button,index) => {

        button.disabled = true;

        if (index === q.answer) {
            button.classList.add("correct");
        }

        if (
            index === selected &&
            selected !== q.answer
        ) {
            button.classList.add("wrong");
        }

    });

    const feedback =
        document.getElementById("feedback");

    feedback.className = "answer-feedback show";

    if (selected === q.answer) {

        feedback.classList.add("feedback-correct");

        feedback.innerHTML =
            `✅ <b>Correct!</b><br><br>${q.explanation}`;

    } else {

        feedback.classList.add("feedback-wrong");

        feedback.innerHTML =
            `❌ <b>Wrong Answer!</b><br><br>
             Correct answer:
             <b>${q.options[q.answer]}</b>
             <br><br>
             ${q.explanation}`;

    }

    if (scroll) {

        feedback.scrollIntoView({
            behavior:"smooth",
            block:"nearest"
        });
    }
}


/* =========================
   QUIZ STATS
========================= */

function updateQuizStats() {

    let correct = 0;

    let wrong = 0;

    userAnswers.forEach((answer,index) => {

        if (answer === null) return;

        if (answer === quizQuestions[index].answer) {
            correct++;
        } else {
            wrong++;
        }

    });

    document.getElementById("correctCount").textContent =
        correct;

    document.getElementById("wrongCount").textContent =
        wrong;
}


/* =========================
   NEXT QUESTION
========================= */

function nextQuestion() {

    if (userAnswers[quizIndex] === null) {

        showToast("Please answer the question first");

        return;
    }

    if (quizIndex < quizQuestions.length - 1) {

        quizIndex++;

        renderQuestion();

    } else {

        finishQuiz();
    }
}


/* =========================
   PREVIOUS
========================= */

function previousQuestion() {

    if (quizIndex > 0) {

        quizIndex--;

        renderQuestion();
    }
}


/* =========================
   FINISH QUIZ
========================= */

function finishQuiz() {

    let correct = 0;

    userAnswers.forEach((answer,index) => {

        if (
            answer !== null &&
            answer === quizQuestions[index].answer
        ) {
            correct++;
        }

    });

    /*
       IMPORTANT:
       Percentage is calculated only from
       the actual correct answers.

       Wrong answers can NEVER produce 100%.
    */

    const percentage =
        Math.round(
            (correct / quizQuestions.length) * 100
        );

    document.getElementById("quizCard").style.display =
        "none";

    document.getElementById("scoreResult").style.display =
        "block";

    document.getElementById("finalScore").textContent =
        percentage + "%";

    document.getElementById("scoreDetails").textContent =
        `${correct} correct out of ${quizQuestions.length} questions.`;

    let message = "";

    if (percentage === 100) {

        message = "🏆 Perfect Score! Excellent coding knowledge.";

    } else if (percentage >= 80) {

        message = "🔥 Excellent! You have strong programming skills.";

    } else if (percentage >= 60) {

        message = "👍 Good job! Keep practicing advanced concepts.";

    } else if (percentage >= 40) {

        message = "📚 Nice attempt! More practice will help.";

    } else {

        message = "💪 Keep learning. You can improve!";
    }

    document.getElementById("scoreMessage").textContent =
        message;

    const oldBest =
        Number(localStorage.getItem("bestScore") || 0);

    if (percentage > oldBest) {

        localStorage.setItem("bestScore",percentage);

        document.getElementById("bestScore").textContent =
            percentage + "%";
    }

    showToast(`Quiz completed: ${percentage}%`);
}

/* =========================
   KEYBOARD SHORTCUTS
========================= */

document.addEventListener("keydown", function(event) {

    if (currentPage !== "quiz") return;

    if (
        event.key >= "1" &&
        event.key <= "4"
    ) {

        const index =
            Number(event.key) - 1;

        const buttons =
            document.querySelectorAll(".option");

        if (buttons[index]) {
            buttons[index].click();
        }
    }

    if (event.key === "ArrowRight") {
        nextQuestion();
    }

    if (event.key === "ArrowLeft") {
        previousQuestion();
    }

});
