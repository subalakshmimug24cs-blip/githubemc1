<script>

/* ================= VARIABLES ================= */

let currentInput = "";
let expressionText = "";
let history = [];
let powerWaiting = false;


/* ================= ELEMENTS ================= */

const resultDisplay =
    document.getElementById("result");

const expressionDisplay =
    document.getElementById("expression");

const historyContent =
    document.getElementById("historyContent");

const scientificPanel =
    document.getElementById("scientificPanel");

const basicBtn =
    document.getElementById("basicBtn");

const scientificBtn =
    document.getElementById("scientificBtn");

const darkBtn =
    document.getElementById("darkBtn");


/* ================= BASIC MODE ================= */

basicBtn.addEventListener("click", () => {

    scientificPanel.classList.remove("show");

    basicBtn.classList.add("active");

    scientificBtn.classList.remove("active");

});


/* ================= SCIENTIFIC MODE ================= */

scientificBtn.addEventListener("click", () => {

    scientificPanel.classList.add("show");

    scientificBtn.classList.add("active");

    basicBtn.classList.remove("active");

});


/* ================= DARK MODE ================= */

darkBtn.addEventListener("click", () => {

    document.body.classList.toggle("dark");

    if (document.body.classList.contains("dark")) {

        darkBtn.textContent = "Light Mode";

    } else {

        darkBtn.textContent = "Dark Mode";

    }

});


/* ================= INPUT ================= */

function inputValue(value) {

    /*
       If power mode is active,
       convert xʸ into ^.
    */

    currentInput += value;

    updateDisplay();
}


/* ================= UPDATE DISPLAY ================= */

function updateDisplay() {

    expressionDisplay.textContent =
        expressionText;

    if (currentInput === "") {

        resultDisplay.textContent = "0";

        return;
    }

    resultDisplay.textContent =
        currentInput
        .replace(/\*/g, "×")
        .replace(/\//g, "÷")
        .replace(/-/g, "−")
        .replace(/\^/g, "ʸ");

}


/* ================= CLEAR ================= */

function clearAll() {

    currentInput = "";

    expressionText = "";

    powerWaiting = false;

    updateDisplay();

}


/* ================= BACKSPACE ================= */

function backspace() {

    currentInput =
        currentInput.slice(0, -1);

    updateDisplay();

}


/* ================= BASIC CALCULATE ================= */

function calculate() {

    if (currentInput.trim() === "") {
        return;
    }

    try {

        let originalExpression =
            currentInput;

        let safeExpression =
            currentInput
            .replace(/×/g, "*")
            .replace(/÷/g, "/")
            .replace(/−/g, "-")
            .replace(/\^/g, "**");

        let result =
            Function(
                '"use strict"; return (' +
                safeExpression +
                ')'
            )();

        if (!Number.isFinite(result)) {
            throw new Error("Invalid");
        }

        result =
            formatNumber(result);

        expressionText =
            originalExpression
            .replace(/\*/g, "×")
            .replace(/\//g, "÷")
            .replace(/-/g, "−")
            .replace(/\^/g, "ʸ")
            + " =";

        currentInput =
            String(result);

        resultDisplay.textContent =
            result;

        addHistory(
            originalExpression,
            result
        );

        powerWaiting = false;

    }

    catch (error) {

        showError();

    }

}


/* ================= SCIENTIFIC CALCULATOR ================= */

function scientificCalculate(type) {

    let value =
        getCurrentNumber();

    if (value === null) {
        return;
    }

    try {

        let result;

        switch (type) {

            case "sin":

                result =
                    Math.sin(
                        value * Math.PI / 180
                    );

                expressionText =
                    "sin(" + value + ")";

                break;


            case "cos":

                result =
                    Math.cos(
                        value * Math.PI / 180
                    );

                expressionText =
                    "cos(" + value + ")";

                break;


            case "tan":

                result =
                    Math.tan(
                        value * Math.PI / 180
                    );

                expressionText =
                    "tan(" + value + ")";

                break;


            case "sqrt":

                if (value < 0) {
                    throw new Error();
                }

                result =
                    Math.sqrt(value);

                expressionText =
                    "√(" + value + ")";

                break;


            case "square":

                result =
                    Math.pow(value, 2);

                expressionText =
                    value + "²";

                break;


            case "log":

                if (value <= 0) {
                    throw new Error();
                }

                result =
                    Math.log10(value);

                expressionText =
                    "log(" + value + ")";

                break;


            case "ln":

                if (value <= 0) {
                    throw new Error();
                }

                result =
                    Math.log(value);

                expressionText =
                    "ln(" + value + ")";

                break;


            case "inverse":

                if (value === 0) {
                    throw new Error();
                }

                result =
                    1 / value;

                expressionText =
                    "1/" + value;

                break;


            case "factorial":

                result =
                    factorial(value);

                expressionText =
                    value + "!";

                break;

        }

        result =
            formatNumber(result);

        currentInput =
            String(result);

        resultDisplay.textContent =
            result;

        addHistory(
            expressionText,
            result
        );

    }

    catch (error) {

        showError();

    }

}


/* ================= POWER ================= */

function powerMode() {

    if (currentInput === "") {
        return;
    }

    currentInput += "^";

    expressionText =
        currentInput
        .replace(/\^/g, "ʸ");

    powerWaiting = true;

    updateDisplay();

}


/* ================= CONSTANTS ================= */

function insertConstant(type) {

    if (type === "pi") {

        currentInput +=
            Math.PI.toString();

    }

    else if (type === "e") {

        currentInput +=
            Math.E.toString();

    }

    updateDisplay();

}


/* ================= GET CURRENT NUMBER ================= */

function getCurrentNumber() {

    if (currentInput === "") {

        let displayValue =
            parseFloat(
                resultDisplay.textContent
            );

        if (Number.isNaN(displayValue)) {
            return null;
        }

        return displayValue;
    }

    /*
       Scientific functions work on
       the last calculated number.
    */

    if (
        currentInput.includes("+") ||
        currentInput.includes("-") ||
        currentInput.includes("*") ||
        currentInput.includes("/") ||
        currentInput.includes("^")
    ) {

        return null
